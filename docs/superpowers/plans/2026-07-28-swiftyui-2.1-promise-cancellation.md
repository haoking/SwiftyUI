# SwiftyUI 2.1 实施计划：Promise 取消语义

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 `Promise` 观察 `Operation` 的取消，使被取消的链条产生一个调用方能看见的错误，而不是静默给出错误结果或永久挂死。

**Architecture:** 取消判定集中在 `completionBlock` —— 链条本来就唯一的推进入口。链条任务与 `always` 任务都挂上感知取消的 `completionBlock`，三种坏表现（陈旧值继续、静默丢失、`value()` 挂死）因此收敛到同一个机制。判定条件是 `isCancelled && !didRun`，不是单看 `isCancelled`。

**Tech Stack:** Swift 6 语言模式、iOS 15、XCTest、Foundation `Operation`/`OperationQueue`。

**设计文档：** `docs/superpowers/specs/2026-07-28-swiftyui-2.1-promise-cancellation-design.md`

```swift
/// 测试里跨线程读写单个值的小盒子。
///
/// @unchecked Sendable 的理由与库内的 ResumeGuard 一致:唯一的可变状态
/// 只经由 lock 保护的临界区读写，编译器无法验证，故为 unchecked。
final class Locked<T>: @unchecked Sendable {
  private let lock = NSLock()
  private var value: T

  init(_ initial: T) { self.value = initial }

  func get() -> T {
    lock.lock()
    defer { lock.unlock() }
    return value
  }

  func set(_ newValue: T) {
    lock.lock()
    defer { lock.unlock() }
    value = newValue
  }
}
```

// MARK: - 取消(2.1)

/// 链条中的任务在开始执行前被取消 → catch 收到 PromiseError.cancelled，
/// 且后续任务不再执行。
///
/// 全程在一个**独立**的线程池上进行:先挂起它，让 operation 排队但不开始，
/// 再按 identifier 精确取消我们自己那一个，然后放行。不碰
/// SwiftyThreadPool.default —— 那是进程级单例，动它会污染别的测试。
func testCancelledTaskFailsChainWithCancelledError() {
  let pool = SwiftyThreadPool()
  pool.stop()

  let secondTaskRan = Locked(false)
  let received = Locked<Error?>(nil)
  let settled = XCTestExpectation(description: "chain settles")

  let promise = Promise<Int>.firstly(pool: pool, with: "first", on: .background) { (update, _) in
    update(1)
  }.then(on: .background) { (_, _) in
    secondTaskRan.set(true)
  }

  promise.catch { error in
    received.set(error)
    settled.fulfill()
  }

  // 此时第一个 operation 已排队在挂起的池里。精确取消它。
  XCTAssertTrue(pool.remove(withIdentifier: "first"), "应当找到并取消排队中的第一个 operation")
  pool.restart()

  wait(for: [settled], timeout: 20)
  XCTAssertEqual(
    received.get() as? PromiseError, .cancelled,
    "被取消的任务必须让链条以 PromiseError.cancelled 失败")
  XCTAssertFalse(secondTaskRan.get(), "链条已因取消而终结，后续任务不得执行")
}

/// 取消发生在执行**期间**时，工作其实已经完成 —— 不得因此丢弃有效结果。
/// 判定条件是 isCancelled && !didRun，不是单看 isCancelled。
func testCancelDuringExecutionDoesNotDiscardCompletedWork() async throws {
  let pool = SwiftyThreadPool()

  let promise = Promise<Int>.firstly(pool: pool, on: .background) { (update, _) in
    update(42)
    // 任务体跑完之后才取消:BlockOperation 的块不检查 isCancelled，
    // 所以这份工作是完整完成的，不该因为一个迟到的 cancel 被丢弃。
    pool.removeAll()
  }

  let value = try await withTestTimeout { try await promise.value() }
  XCTAssertEqual(value, 42, "取消发生在执行期间时工作已完成，结果不得被丢弃")
}
```

```swift
/// Promise 自身产生的错误（区别于任务体抛出的业务错误）。
public enum PromiseError: Error, Sendable, Equatable {
  /// 支撑该任务的 Operation 在开始执行前被取消。
  ///
  /// 最常见的来源是 `SwiftyThreadPool` 收到内存告警后调用 `removeAll()`。
  case cancelled

  /// 链条正常终结，但没有任何任务调用过 `update(...)`，因此没有值可返回。
  case noValue
}

/// 记录一个 operation 的执行块是否真的运行过。
///
/// 存在的理由:BlockOperation 的执行块不检查 isCancelled，所以取消若发生在
/// 执行**期间**，工作其实已经完成 —— 此时把链条判为取消会丢弃一个有效结果。
/// 判定条件因此是 `isCancelled && !didRun`，而不是单看 isCancelled。
///
/// Foundation 保证 completionBlock 在 operation 结束之后运行，因此在
/// completionBlock 里读到的 didRun 一定反映了执行块是否跑过。
///
/// @unchecked Sendable 的理由与 ResumeGuard 一致:唯一的可变状态只经由
/// lock 保护的临界区读写，编译器无法验证，故为 unchecked。
private final class RunFlag: @unchecked Sendable {
  private let lock = NSLock()
  private var value = false

  func markRun() {
    lock.lock()
    defer { lock.unlock() }
    value = true
  }

  var didRun: Bool {
    lock.lock()
    defer { lock.unlock() }
    return value
  }
}
```

```swift
private struct PromiseTask {
  var identifier: String?
  var perform: ThreadPerform
  var op: Operation
  /// 与本任务的 op 一一对应，由 op 的执行块置位。
  let runFlag: RunFlag
  /// 仅 value() 传入:该 always 任务被取消时用来抛错恢复 continuation。
  /// 用户经 always(_:) 追加的任务不设此项 —— 链条此时已终结，错误无处可去。
  let onCancelled: (@Sendable () -> Void)?

  private init(
    _ identifier: String? = nil, _ perform: ThreadPerform = .background, _ op: Operation,
    _ runFlag: RunFlag, _ onCancelled: (@Sendable () -> Void)?
  ) {
    self.identifier = identifier
    self.perform = perform
    self.op = op
    self.runFlag = runFlag
    self.onCancelled = onCancelled
  }

  static func create(
    _ identifier: String? = nil, _ perform: ThreadPerform = .background, _ op: Operation,
    runFlag: RunFlag, onCancelled: (@Sendable () -> Void)? = nil
  ) -> PromiseTask {
    return PromiseTask(identifier, perform, op, runFlag, onCancelled)
  }

  /// 池由调用方传入而不是直接取 SwiftyThreadPool.default:取消语义必须能在
  /// 一个与其他测试隔离的池上验证。对外的公开 API 一律仍走共享单例。
  func start(on pool: SwiftyThreadPool) {
    switch perform {
    case .main:
      OperationQueue.main.addOperation(op)
    case .background:
      pool.add(op, withIdentifier: identifier)
    }
  }
}
```

- [ ] **Step 7: 让两个 `operationBuild` 连同 flag 一起返回**

```swift
private func operationBuild(_ work: (@Sendable () throws -> Void)?) -> (
  op: Operation, flag: RunFlag
) {
  let flag = RunFlag()
  // 这里不持有 self:链条运行期间的强引用由 fire() 在投递时挂到
  // operation 的 completionBlock 上,详见 fire()
  let op = BlockOperation { [weak self] in
    // 必须是块内第一句:哪怕 self 已经没了,"这个块运行过"也是事实,
    // 否则 completionBlock 会把它误判为取消。
    flag.markRun()
    guard let self = self else { return }
    do {
      try work?()
    } catch {
      self.markError(error)
    }
  }
  return (op, flag)
}

private func operationBuildT(
  _ work: @escaping @Sendable (_ update: (T?) -> Void, _ value: T?) throws -> Void
) -> (op: Operation, flag: RunFlag) {
  let flag = RunFlag()
  let op = BlockOperation { [weak self] in
    flag.markRun()
    guard let self = self else { return }
    do {
      try work(self.update, self.currentValue)
    } catch {
      self.markError(error)
    }
  }
  return (op, flag)
}
```

- [ ] **Step 10: `fire()` 的两个 `task.start()` 都要传池，`.run` 分支加取消判定**

```swift
switch next {
case .run(let task):
  // 已投递的 operation 是链条运行期间对 self 的唯一强引用:调用方不持有
  // Promise 时链条也能跑完。它不构成循环引用——task 在上面的临界区里已被
  // 移出 tasks,operation 完成后由队列释放,强引用随之解除。
  //
  // 只捕获 flag,**不要**捕获 task:task 持有 op,把它捕进 op 自己的
  // completionBlock 会造成 op → completionBlock → task → op 的环 ——
  // 正是下面 [weak op] 想避免的东西。RunFlag 不引用 op,捕获它是安全的。
  let flag = task.runFlag
  task.op.completionBlock = { [weak op = task.op] in
    // isCancelled 单独成立不够:取消若发生在执行期间,工作其实已完成,
    // 此时判为取消会丢弃一个有效结果(见 RunFlag 的说明)。
    if op?.isCancelled == true && !flag.didRun {
      // markError 置 isError,紧接着的 fire() 会走 isError 分支丢弃剩余
      // 任务并终结 —— 复用既有的错误终结路径,不新增状态。
      self.markError(PromiseError.cancelled)
    }
    self.fire()
  }
  task.start(on: pool)
case .finish(let pending):
  // always 在链条终结之后才执行(B11)
  for task in pending {
    task.start(on: pool)
  }
case .nothing:
  break
}
```

/// markError 会经 internalErrorHandler 恢复 continuation，测试不修也会绿。
/// 必须让链条**成功**终结、只取消 value() 那个 always 任务，才测到真东西。
///

/// 测试会挂住整个测试进程而不是失败 —— 那正是要防的东西。
func testAsyncValueThrowsWhenItsAlwaysTaskIsCancelled() async throws {
  let pool = SwiftyThreadPool()
  pool.stop()  // always 任务会排队但不开始

  let promise = Promise<Int>.firstly(pool: pool, on: .main) { (update, _) in
    update(1)
  }

  let waiter = Task { try await withTestTimeout(10) { try await promise.value() } }

  // 等链条把 .main 上的任务跑完并把 always 投递到(挂起的)池里。
  // 轮询内部状态而不是 sleep 一个魔法时长。
  let deadline = Date().addingTimeInterval(5)

    try await Task.sleep(nanoseconds: 10_000_000)

  pool.restart()

    _ = try await waiter.value
    XCTFail("always 任务被取消时 value() 必须抛错")
  } catch let error as PromiseError {
    XCTAssertEqual(error, .cancelled, "必须是 PromiseError.cancelled")

  background: UIColor, primary: UIColor, secondary: UIColor, detail: UIColor
) {
  let promise = Promise<(UIColor, UIColor, UIColor, UIColor)>.firstly { [weak self] (update, _) in
    guard let strongSelf = self else { return }
    update(strongSelf.getColors())
  }
  guard let colors = try await promise.value() else {
    // value() 返回 nil 意味着没有任务调用过 update(...)，在这里只可能是
    // 上面那个 weak self 守卫失败。不强解包 —— B1/B19 都是这么来的。
    throw PromiseError.noValue
  }
  return (colors.0, colors.1, colors.2, colors.3)
}
````

echo "使用模拟器:${DEVICE_NAME:-未知} ($UDID)"
